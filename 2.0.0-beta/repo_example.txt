<!--
    MacWave Repository Format Example
    This file is a reference for understanding the repoindex.txt syntax.
    It is not used directly by MacWave.
    Location: ~/.local/macwave/repo_example.txt
-->

<!-- ==================== 1. Package Name List ==================== -->
<!-- 
    The def repo_dict block lists all available package names.
    The parser uses this list to quickly iterate over all packages.
-->
def repo_dict:
    "ldid"
    "machox"
    "palera1n"
    "test_001"

<!-- ==================== 2. URL Variable Definitions ==================== -->
<!-- 
    The {repo_url block defines download URLs as variables.
    This allows binary_url fields to reference these variables using $name,
    avoiding repeated long URLs.
-->
{repo_url:
    let "ldid" = "https://raw.githubusercontent.com/ProcursusTeam/ldid/main/ldid"
    let "machox" = "https://github.com/Sha0huaZhang/MachOX/releases/latest/download/machox"
    let "palera1n" = "https://github.com/palera1n/palera1n/releases/latest/download/palera1n-macos-universal"
    let "test_001" = "https://raw.githubusercontent.com/MacWaveOrg/MacWave/main/test/test_001"
}

<!-- ==================== 3. Package Details ==================== -->
<!-- 
    The [package_info block contains full metadata for each package.
    Each entry includes version, architecture, SHA256 checksum, and binary_url.
    The binary_url field can use $variable_name to reference URLs from repo_url.
-->
[package_info:
    "ldid":
        version: "2.1.5-procursus7"
        arch: "any"
        sha256: "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
        binary_url: "$ldid"

    "machox":
        version: "1.0.0"
        arch: "any"
        sha256: "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
        binary_url: "$machox"

    "palera1n":
        version: "2.0.0-beta.9"
        arch: "any"
        sha256: "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
        binary_url: "$palera1n"

    "test_001":
        version: "1.0.0"
        arch: "any"
        sha256: "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"
        binary_url: "$test_001"
]
